/* Canvas renderer uses the server's 1000×500 playing coordinates. * A 100-unit surround provides space for the rails and the cue. * It never simulates outcomes: authoritative frames come from the server. */ (() => { const W = 1200, H = 700, R = 14, TAU = Math.PI * 2; // The expanded target radius gives the location of the cue-ball centre at impact. function ray(cue, balls, angle) { const ux = Math.cos(angle), uy = Math.sin(angle); const candidates = []; if (ux > 1e-9) candidates.push({ t: (bounds.right - cue.x) / ux, axis: "x" }); if (ux < -1e-9) candidates.push({ t: (bounds.left - cue.x) / ux, axis: "x" }); if (uy > 1e-9) candidates.push({ t: (bounds.bottom - cue.y) / uy, axis: "y" }); if (uy < -1e-9) candidates.push({ t: (bounds.top - cue.y) / uy, axis: "y" }); const wall = candidates .filter((v) => v.t >= 0) .sort((a, b) => a.t - b.t)[0] || { t: 0, axis: "x" }; let distance = wall.t, hit = null; for (const ball of balls) { if (!ball.id || ball.pocketed) continue; const dx = ball.x - cue.x, dy = ball.y - cue.y, projection = dx * ux + dy * uy; if (projection <= 0) continue; const side = dx * dx + dy * dy - projection * projection; if (side > (2 * R) ** 2) continue; const t = projection - Math.sqrt(Math.max(0, (2 * R) ** 2 - side)); if (t >= 0 && t < distance) { distance = t; hit = ball; } } return { x: cue.x + ux * distance, y: cue.y + uy * distance, hit, axis: wall.axis, };
